await initHydra()


let sc = "<[C]!2 [D]!4 [Ab2]!2>:<[pentatonic | minor | major]!2>"

let seg = "[16 | 8? | 3?]"
let fil = sine.range(200,10000)

////////////////////////////////////////////////////////////////////////////////////Hydra\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\

osc(H(seg), -0.1,  H(seg)).rotate(H(fil)).saturate(0.4).luma(0.9, 0.4)
.kaleid(H(seg)).out()
////////////////////////////////////////////////////////////////////////////////Music\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\

setcpm(140/4)

$: n(rand.range(0, 8)).segment(4).scale(sc).sound("sine").gain(2)
$: n(rand.range(0, 
4)).segment("2?0.3").scale(sc).sound("sbd").gain(0.3).mask("<0/4 [1 | 0] 
 >").delay("<0 .25 .5 1>")
$: n(rand.range(0, 5)).segment(8).scale(sc).transpose(12).sound("tri")
$: n(rand.range(0, 5)).segment(seg).scale(sc).transpose("[12?0.1 12?0.3 
0?0.2 
-12?0.2]").sound("sqr").attack("0.025").release("0.025").lpf(sine.range(500, 
15000)).gain(0.2).room(0.6)

//drum
$: s("- sd - sd").gain(0.4)

$: s("[alesissr16_bd hh alesissr16_bd hh alesissr16_bd hh alesissr16_bd 
hh?]*[1 | 0.5]").gain(1.3).lpf(fil).jux(iter(4))
